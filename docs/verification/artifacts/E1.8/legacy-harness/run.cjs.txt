// This wrapper NEVER starts, stops, creates or drops a database/container.
const path = require('node:path');
const fs = require('node:fs');
const { spawnSync } = require('node:child_process');
const { createRequire } = require('node:module');
const { guard, ROOT } = require('./guard.cjs');
const OLD = path.join(__dirname, 'source');
const phase = process.argv[2];
if (phase === 'inspect') {
  const config = JSON.parse(fs.readFileSync(path.join(OLD, 'tsconfig.json'), 'utf8'));
  const configAlias = path.resolve(OLD, config.compilerOptions.paths['@payload-config'][0]);
  const sourceAlias = path.resolve(OLD, config.compilerOptions.paths['@/*'][0].replace('*', 'config/server-env.ts'));
  if (configAlias !== path.join(OLD, 'src/payload.config.ts') || sourceAlias !== path.join(OLD, 'src/config/server-env.ts')) throw new Error('Old aliases escaped archived source.');
  if (!fs.existsSync(sourceAlias) || fs.existsSync(path.join(OLD, '.git')) || fs.existsSync(path.join(OLD, 'node_modules'))) throw new Error('Invalid isolated source archive.');
  const dependency = createRequire(path.join(OLD, 'package.json')).resolve('payload');
  if (!dependency.toLowerCase().startsWith(path.join(ROOT, 'node_modules').toLowerCase())) throw new Error('Dependencies must resolve from the established node_modules.');
  console.log(JSON.stringify({ kind: 'STATIC_ONLY', configAlias, sourceAlias, payloadDependency: dependency, archiveBaseline: '0fa495864cfefcf3489856d66444b81fdf1f730e' }));
  process.exit(0);
}
guard(); // Before Payload CLI loadEnv, tsx, config or app imports.
const old = phase === 'old-migrate' || phase === 'old-seed';
const allowed = ['old-migrate', 'old-seed', 'current-migrate', 'captured', 'bootstrap', 'verify'];
if (!allowed.includes(phase)) throw new Error('Choose an explicitly documented phase.');
const selected = old ? OLD : ROOT;
const env = { ...process.env, TSX_TSCONFIG_PATH: path.join(selected, 'tsconfig.json'), PAYLOAD_CONFIG_PATH: path.join(selected, 'src/payload.config.ts') };
const cli = path.join(ROOT, 'node_modules/payload/bin.js');
const args = ['--conditions=react-server', cli];
if (phase.endsWith('migrate')) args.push('migrate');
else if (phase === 'old-seed') args.push('run', path.join(__dirname, 'seed.ts'));
else if (phase === 'bootstrap') args.push('run', path.join(ROOT, 'src/config/bootstrap-publications.cli.ts'));
else args.push('run', path.join(__dirname, 'verify.ts'), phase);
const result = spawnSync(process.execPath, args, { cwd: ROOT, env, stdio: 'inherit', windowsHide: true });
if (result.error) throw new Error('Child process could not start.');
process.exit(result.status ?? 1);
