const fs = require('node:fs');
const path = require('node:path');
const ts = require('../../node_modules/typescript');
for (const name of ['seed.ts', 'verify.ts']) {
  const file = path.join(__dirname, name);
  const result = ts.transpileModule(fs.readFileSync(file, 'utf8'), { fileName: file, reportDiagnostics: true, compilerOptions: { target: ts.ScriptTarget.ES2022, module: ts.ModuleKind.ESNext } });
  if ((result.diagnostics || []).some((item) => item.category === ts.DiagnosticCategory.Error)) throw new Error('Prepared TypeScript syntax failed.');
}
console.log('STATIC_ONLY: prepared TypeScript transpilation passed; no module imported or initializer run.');
